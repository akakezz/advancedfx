cmake_minimum_required(VERSION 3.24)
project(tf2_camera_tests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
enable_testing()
add_executable(cinematic_tests test_cinematic.cpp ../../shared/CamPath.cpp ../../shared/AfxMath.cpp ../../shared/MirvCampath.cpp)
target_include_directories(cinematic_tests PRIVATE ../.. ../../AfxHookSource)
if(NOT MSVC)
    target_compile_definitions(cinematic_tests PRIVATE abstract=)
    target_compile_options(cinematic_tests PRIVATE -include ${CMAKE_CURRENT_SOURCE_DIR}/msvc_crt_compat.h)
endif()
add_test(NAME camera_core COMMAND cinematic_tests)

if(UNIX AND DEFINED AFX_IMGUI_SOURCE_DIR)
    find_package(Python3 REQUIRED COMPONENTS Interpreter)
    add_test(NAME camera_overlay_gui COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/test_overlay.py" --imgui-dir "${AFX_IMGUI_SOURCE_DIR}" --build-dir "${CMAKE_CURRENT_BINARY_DIR}/overlay-gui")
endif()

add_executable(sequencer_tests test_sequencer.cpp ../../shared/PlayerRepair.cpp ../../shared/CommandSystem.cpp ../../shared/DemoTimeline.cpp ../../shared/AfxMath.cpp ../../shared/AfxConsole.cpp)
target_include_directories(sequencer_tests PRIVATE ../.. ../../AfxHookSource)
if(NOT MSVC)
    target_compile_definitions(sequencer_tests PRIVATE abstract=)
    target_compile_options(sequencer_tests PRIVATE -include ${CMAKE_CURRENT_SOURCE_DIR}/msvc_crt_compat.h)
endif()
add_test(NAME sequencer_core COMMAND sequencer_tests)

add_executable(scene_cache_tests scene_cache_test.cpp ../../shared/AfxMath.cpp)
if(NOT MSVC)
    target_compile_definitions(scene_cache_tests PRIVATE abstract=)
    target_compile_options(scene_cache_tests PRIVATE -include ${CMAKE_CURRENT_SOURCE_DIR}/msvc_crt_compat.h)
endif()
target_include_directories(scene_cache_tests PRIVATE ../.. ../../AfxHookSource)
add_test(NAME scene_cache_core COMMAND scene_cache_tests)

if(UNIX)
    find_package(Python3 REQUIRED COMPONENTS Interpreter)
    add_test(NAME scene_pose_native COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/test_scene_pose.py" --build-dir "${CMAKE_CURRENT_BINARY_DIR}/scene-pose-native")
    add_test(NAME overlay_input_native COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/test_overlay_input.py" --build-dir "${CMAKE_CURRENT_BINARY_DIR}/overlay-input-native")
endif()
